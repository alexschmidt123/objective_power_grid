\begin{abstract}
This paper studies sequential Bayesian experimental design for uncertain
power-grid dynamics. At each stage, a policy selects a probing bus and duration,
observes a noisy frequency response, and updates uncertainty over
machine-specific parameters. We distinguish parameter-oriented expected
information gain (EIG) from objective-driven mean objective cost of uncertainty
(MOCU) for a downstream safety-aware control decision. Random, fixed, myopic,
deep adaptive design (DAD), and reinforcement-learning sequential Bayesian
optimal experimental design (RL-sBOED), and Step-DAD are evaluated under common
simulation banks and held-out protocols. A susceptible--infected--recovered (SIR) model
provides an implementation benchmark; IEEE 9-bus, IEEE 14-bus, and IEEE 30-bus
systems assess the power-grid application and scalability. \todo{Add final
quantitative findings.}
\end{abstract}

\section{Introduction}
\IEEEPARstart{U}{ncertainty} in power-system inertia and fast frequency response
complicates frequency-security assessment and control selection. Controlled
probing can reduce uncertainty, but an experiment must determine where to probe,
how long to probe, and whether later probes should depend on earlier
observations. \todo{Add motivation and citations.}

EIG values parameter learning, but parameter information does not necessarily
improve a final control decision. We therefore also study MOCU, which values an
experiment by its reduction of expected decision penalty.

The contributions are:
\begin{itemize}
  \item sequential bus--duration BOED for uncertain power-grid dynamics;
  \item a safety-aware MOCU objective linking probing to terminal control;
  \item a matched comparison of random, fixed, myopic, DAD, RL-sBOED, and
  Step-DAD; and
  \item SIR, IEEE9, IEEE14, and IEEE30 studies of objective choice, adaptation,

\section{Problem Formulation}

\subsection{Uncertain Power-Grid Dynamics}
For retained dynamic machine $i$,
\begin{align}
\dot{\delta}_i(t)&=\omega_i(t),\\
M_i\dot{\omega}_i(t)&=P_{m,i}-\sum_jB_{ij}\sin(\delta_i-\delta_j)
-\left(\frac{K_i}{2\pi}+D_i\right)\omega_i+p_i(t;d).
\end{align}
The uncertain vector is
\begin{equation}
\boldsymbol{\theta}=(M_1,\ldots,M_G,K_1,\ldots,K_G).
\end{equation}
Only dynamic machines have $M_i$ and $K_i$. IEEE14 is Kron-reduced onto
machines at physical buses 1, 2, 3, 6, and 8. \todo{State and justify priors.}

\subsection{Sequential Probing Design}
At stage $t$,
\begin{equation}
d_t=(b_t,\tau_t)\in\mathcal D,
\end{equation}
where $b_t$ is a physical probing bus and $\tau_t$ is one of six durations.
IEEE9 has $9\times6=54$ actions and IEEE14 has $14\times6=84$ actions. The
policy selects $d_t=\pi_t(h_{t-1})$ from the previous history.

\subsection{Observation and Belief Update}
\begin{equation}
\mathbf y_t=g(\boldsymbol{\theta},d_t)+\boldsymbol{\epsilon}_t,
\qquad \boldsymbol{\epsilon}_t\sim\mathcal N(\mathbf0,\sigma^2\mathbf I).
\end{equation}
Power-grid probes retain five frequency-response observations. Bayesian
updating gives
\begin{equation}
p(\boldsymbol{\theta}\mid h_t)\propto
p(\mathbf y_t\mid\boldsymbol{\theta},d_t)
p(\boldsymbol{\theta}\mid h_{t-1}).
\end{equation}

\subsection{EIG and MOCU Objectives}
\begin{equation}
\operatorname{EIG}(d\mid h)=\E_{\mathbf y\mid d,h}
\left[D_{\mathrm{KL}}\left(p(\boldsymbol{\theta}\mid h,d,\mathbf y)
\Vert p(\boldsymbol{\theta}\mid h)\right)\right].
\end{equation}
Let $u\in\mathcal U$ be a terminal control and $C(u,\boldsymbol{\theta})$ its
safety-aware cost. Define
\begin{align}
u^*(\boldsymbol{\theta})&=\arg\min_uC(u,\boldsymbol{\theta}),\\
u^*(h_T)&=\arg\min_u\E_{\boldsymbol{\theta}\mid h_T}
[C(u,\boldsymbol{\theta})],\\
\ocu(\boldsymbol{\theta},h_T)&=C(u^*(h_T),\boldsymbol{\theta})
-C(u^*(\boldsymbol{\theta}),\boldsymbol{\theta}),\\
\mocu(h_T)&=\E_{\boldsymbol{\theta}\mid h_T}
[\ocu(\boldsymbol{\theta},h_T)].
\end{align}
The objective-driven policy minimizes $\E_\pi[\mocu(h_T)]$; lower is better.
