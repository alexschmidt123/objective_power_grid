    49	def primary_backward(benchmark,model,truth,bank,noise,seed,microbatch=128):
    50	    """Exact effective-batch objective with global detached balance frequency.
    51	
    52	    The cheap truth-only pass fixes global assignment frequencies. Microbatch
    53	    sPCE and balance gradients sum to the original B-wide loss; no optimizer
    54	    update occurs between chunks. This is not an L/B reduction.
    55	    """
    56	    batch=len(truth);contrasts=bank.shape[1];calls=benchmark.calls
    57	    with torch.no_grad():
    58	        first=rollout(benchmark,model,30,batch,seed,0,truth=truth,noise=noise,prior_particles=truth.new_empty(batch,0,4))
    59	        ids=torch.cat([x[1] for x in model._gate_cache]);frequency=torch.nn.functional.one_hot(ids,7).sum(1).double().mean(0)*(7/3)
    60	        del first,ids
    61	    model._gate_cache=[];mean=0.;saturated=0;balance_value=0.
    62	    for start in range(0,batch,microbatch):
    63	        n=min(microbatch,batch-start)
    64	        result=rollout(benchmark,model,30,n,seed,contrasts,truth=truth[start:start+n],noise=noise[start:start+n],prior_particles=bank[start:start+n])
    65	        info=result['info'][:,-1];prob=torch.cat([x[0] for x in model._gate_cache]).mean(0)
    66	        balance=model.balance_factor*(frequency.detach()*prob).sum()
    67	        loss=(-info.mean()+balance)*(n/batch);loss.backward()
    68	        mean+=float(info.detach().sum())/batch;saturated+=int((info.detach()>math.log(contrasts+1)-.1).sum());balance_value+=float(balance.detach())*(n/batch)
    69	        model._gate_cache=[];del result,info,prob,balance,loss
    70	    return dict(main_spce=mean,saturation_fraction=saturated/batch,balance=balance_value,
    71	        main_likelihood_particle_stages=benchmark.calls-calls,extra_truth_only_replay_trajectories=batch)

    73	def auxiliary_backward(benchmark,model,truth,noise,seed,cfg,stage_override=None):
    74	    g=torch.Generator(device=truth.device).manual_seed(seed)
    75	    index=int(torch.randint(0,len(truth),(1,),generator=g,device=truth.device))
    76	    stage=int(torch.randint(0,30,(1,),generator=g,device=truth.device)) if stage_override is None else stage_override
    77	    estimate,stats=conditional_auxiliary(benchmark,model,truth[index:index+1],noise[index:index+1],stage,seed+2000003,cfg['conditional_negatives'],cfg['posterior_particles'],cfg['posterior_moves'])
    78	    gradients=torch.autograd.grad(cfg['alpha_cond']*estimate,tuple(model.parameters()),retain_graph=True,allow_unused=True)
    79	    groups={};total=0.
    80	    for (name,p),grad in zip(model.named_parameters(),gradients):
    81	        if grad is None:continue
    82	        if not torch.isfinite(grad).all():raise FloatingPointError('Nonfinite conditional policy gradient')
    83	        norm=float(grad.square().sum());total+=norm
    84	        key='head' if name.startswith('emitter') else '.'.join(name.split('.')[:2]) if name.startswith('routed.') else name.split('.')[0]
    85	        groups[key]=groups.get(key,0.)+norm
    86	    (-cfg['alpha_cond']*estimate).backward();model._gate_cache=[]
    87	    stats.update(alpha=cfg['alpha_cond'],auxiliary_gradient_norm=math.sqrt(total),
    88	        auxiliary_gradient_groups={k:math.sqrt(v) for k,v in groups.items()},
    89	        sampling_distribution_gradient='STOPPED; biased auxiliary gradient, original full-sPCE remains anchor')
    90	    return stats

    18	def sample_prefix(benchmark,actions,observations,negatives,seed,particles=512,moves=8):
    19	    if benchmark.name!='location_finding':raise ValueError('Only the verified location likelihood is adapted')
    20	    if len(actions)!=len(observations) or negatives>particles:raise ValueError('Invalid prefix/particle dimensions')
    21	    if any(len(a)!=1 or len(y)!=1 for a,y in zip(actions,observations)):raise ValueError('One independently sampled history per pool')
    22	    started=time.monotonic();calls=benchmark.calls
    23	    rng=torch.Generator(device=torch.empty(0).device).manual_seed(int(seed))
    24	    smc=AuditedSMC(benchmark,1,particles,rng,moves=moves)
    25	    for action,y in zip(actions,observations):smc.update(action.detach(),y.detach())
    26	    theta=smc.theta[0];unique=float(torch.unique(theta,dim=0).shape[0]/particles)
    27	    acceptance=min((float(x['mh_acceptance'][0]) for x in smc.diagnostics),default=1.)
    28	    pre_ess=min((x['minimum_ess']/particles for x in smc.diagnostics),default=1.)
    29	    finite=bool(torch.isfinite(theta).all());quality_ok=finite and unique>=.5 and acceptance>=.005 and pre_ess>=.79
    30	    selected=torch.randperm(particles,generator=rng,device=theta.device)[:negatives]
    31	    draws=theta[selected].detach()
    32	    q=PosteriorQuality(len(actions),renewed=True,exact=False,validated=quality_ok,ess_fraction=pre_ess,unique_fraction=unique,
    33	        mixing_checked=acceptance>=.005,symmetry_checked=True,proposal_weights_checked=True,
    34	        sampling_gradient='SMC/resampling and history-conditioning distribution detached; heuristic policy gradient')
    35	    require_quality(q,len(actions))
    36	    return draws,q,dict(status='PASS_EMPIRICAL_POOL_CHECKS',conditioning_observations=len(actions),
    37	        particle_count=particles,negative_count=negatives,pre_resampling_minimum_ess_fraction=pre_ess,
    38	        final_ess_is_uniform_after_resampling=True,final_unique_fraction=unique,
    39	        minimum_mh_acceptance=acceptance,resampling_unique_minimum=min(smc.resampling_unique,default=1.),
    40	        symmetry_label_fraction=float((theta[:,0]<theta[:,2]).double().mean()),
    41	        likelihood_particle_stages=benchmark.calls-calls,posterior_proposals=particles*moves*sum(x['tempering_steps'] for x in smc.diagnostics),
    42	        seconds=time.monotonic()-started,exact=False,independent_draws_certified=False,
    43	        estimate_type='APPROXIMATE_POSTERIOR_HEURISTIC_NOT_A_CERTIFIED_LOWER_BOUND'),theta

    45	def conditional_auxiliary(benchmark,policy,truth,noise,stage,seed,negatives=128,particles=512,moves=8):
    46	    """One uniform time/sample yields T*C, an unbiased stage-sum *selection*.
    47	
    48	    Inference and selection are separate from the primary RNG. Only the
    49	    prefix[:stage] is supplied to SMC. No y_stage or truth enters inference.
    50	    """
    51	    from src.research.engine import rollout
    52	    batch=len(truth)
    53	    if batch!=1:raise ValueError('The frozen first candidate uses one auxiliary case/update')
    54	    start=time.monotonic();calls=benchmark.calls
    55	    result=rollout(benchmark,policy,policy.horizon,batch,seed,0,truth=truth,noise=noise,prior_particles=truth.new_empty(batch,0,4))
    56	    a=[x.detach() for x in result['actions'][:stage]];y=[x.detach() for x in result['observations'][:stage]]
    57	    # The validated CPU sampler is used verbatim; all inference wall time is
    58	    # charged while the GPU lease remains held. Only small draws return to GPU.
    59	    with torch.device('cpu'):
    60	        draws,q,stats,_=sample_prefix(benchmark,[x.cpu() for x in a],[x.cpu() for x in y],negatives,seed+15485863,particles,moves)
    61	    draws=draws.to(truth.device)
    62	    action=result['actions'][stage];observation=result['observations'][stage]
    63	    theta=torch.cat((truth[:,None],draws[None]),1)
    64	    mean,_=benchmark.forward(theta,benchmark.initial(theta.shape[:2]),action[:,None].expand(-1,negatives+1,-1))
    65	    ll=benchmark.log_likelihood(observation[:,None],mean)
    66	    c=ll[:,0]-torch.logsumexp(ll,-1)+math.log(negatives+1)
    67	    if not torch.isfinite(c).all():raise FloatingPointError('Nonfinite V5-A conditional score')
    68	    estimate=policy.horizon*c.mean()
    69	    stats.update(stage=stage,conditional_score=float(c.detach().mean()),stage_sum_estimate=float(estimate.detach()),
    70	        stage_sampling_probability=1/policy.horizon,stage_sampling_weight=policy.horizon,
    71	        total_auxiliary_likelihood_particle_stages=benchmark.calls-calls,
    72	        seconds=time.monotonic()-start,extra_primary_replay_trajectories=1)
    73	    return estimate,stats
